    split_unknown = [r for r in unknown_rows if r[2] == split]

    balanced_unknown.extend(split_unknown[:target])

import torch
import torchaudio
import torch.nn.functional as F

SAMPLE_RATE = 16000  # this dataset's real fixed rate, confirmed in README
N_MELS = 40

mel_transform = torchaudio.transforms.MelSpectrogram(
    sample_rate=SAMPLE_RATE, n_fft=480, hop_length=160, n_mels=N_MELS
)

def load_fixed_waveform(path, label):
    waveform, sr = torchaudio.load(path)
    assert sr == SAMPLE_RATE, f"unexpected sample rate: {sr}"

    if label == "silence":
        # noise files are much longer than 1s -> cut a random 1-second window
        total_len = waveform.shape[1]
        start = random.randint(0, total_len - SAMPLE_RATE)
        waveform = waveform[:, start:start + SAMPLE_RATE]
    else:
        length = waveform.shape[1]
        if length < SAMPLE_RATE:
            waveform = F.pad(waveform, (0, SAMPLE_RATE - length))
        elif length > SAMPLE_RATE:
            waveform = waveform[:, :SAMPLE_RATE]

    return waveform

def to_log_mel(waveform):
    mel = mel_transform(waveform)          # power mel-spectrogram
    log_mel = torch.log(mel + 1e-6)        # log-mel, avoids log(0)
    return log_mel

import torch.nn as nn

class SpeechCNN(nn.Module):
    def __init__(self, num_classes=12):
        super().__init__()
        self.conv = nn.Sequential(
            nn.Conv2d(1, 16, kernel_size=3, padding=1),
            nn.BatchNorm2d(16),
            nn.ReLU(),
            nn.MaxPool2d(2),          # 40x101 -> 20x50

            nn.Conv2d(16, 32, kernel_size=3, padding=1),
            nn.BatchNorm2d(32),
            nn.ReLU(),
            nn.MaxPool2d(2),          # 20x50 -> 10x25

            nn.Conv2d(32, 64, kernel_size=3, padding=1),
            nn.BatchNorm2d(64),
            nn.ReLU(),
            nn.MaxPool2d(2),          # 10x25 -> 5x12
        )
        self.classifier = nn.Sequential(
            nn.Flatten(),
            nn.Linear(64 * 5 * 12, 128),
            nn.ReLU(),
            nn.Dropout(0.3),
            nn.Linear(128, num_classes),
        )

    def forward(self, x):
        x = self.conv(x)
        x = self.classifier(x)
        return x

import time

device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
print("device:", device)

model = SpeechCNN(num_classes=len(LABELS)).to(device)
criterion = nn.CrossEntropyLoss()
optimizer = torch.optim.Adam(model.parameters(), lr=1e-3)

def run_epoch(loader, train=True):
    model.train() if train else model.eval()
    total_loss, correct, total = 0.0, 0, 0
    with torch.set_grad_enabled(train):
        for specs, labels in loader:
            specs, labels = specs.to(device), labels.to(device)
            if train:
                optimizer.zero_grad()
            outputs = model(specs)
            loss = criterion(outputs, labels)
            if train:
                loss.backward()
                optimizer.step()
            total_loss += loss.item() * specs.size(0)
            correct += (outputs.argmax(1) == labels).sum().item()
            total += specs.size(0)
    return total_loss / total, correct / total

start = time.time()
train_loss, train_acc = run_epoch(train_loader, train=True)

random.seed(42)
DATA_DIR = "data/archive"

# --- rebuild file list + splits ---

                   if os.path.isdir(os.path.join(DATA_DIR, d)) and d != "_background_noise_"])

CORE_WORDS = ["yes", "no", "up", "down", "left", "right", "on", "off", "stop", "go"]

with open(os.path.join(DATA_DIR, "validation_list.txt")) as f:
    val_files = set(line.strip() for line in f)
with open(os.path.join(DATA_DIR, "testing_list.txt")) as f:
    test_files = set(line.strip() for line in f)

def label_for(word):
    return word if word in CORE_WORDS else "unknown"

rows = []
for c in classes:
    for path in glob.glob(os.path.join(DATA_DIR, c, "*.wav")):
        rel = os.path.join(c, os.path.basename(path))
        split = "val" if rel in val_files else "test" if rel in test_files else "train"
        rows.append((path, label_for(c), split))

# --- balance unknown + add silence ---
core_rows = [r for r in rows if r[1] != "unknown"]
unknown_rows = [r for r in rows if r[1] == "unknown"]

target_per_split = {"train": 3700, "val": 400, "test": 400}
balanced_unknown = []
for split, target in target_per_split.items():
    split_unknown = [r for r in unknown_rows if r[2] == split]
    random.shuffle(split_unknown)
    balanced_unknown.extend(split_unknown[:target])

NOISE_DIR = os.path.join(DATA_DIR, "_background_noise_")
noise_files = glob.glob(os.path.join(NOISE_DIR, "*.wav"))

silence_rows = []

for split, n in silence_target.items():

# --- audio preprocessing ---
SAMPLE_RATE = 16000
N_MELS = 40
mel_transform = torchaudio.transforms.MelSpectrogram(
    sample_rate=SAMPLE_RATE, n_fft=480, hop_length=160, n_mels=N_MELS
)

def load_fixed_waveform(path, label):
    waveform, sr = torchaudio.load(path)
    assert sr == SAMPLE_RATE
    if label == "silence":
        total_len = waveform.shape[1]
        start = random.randint(0, total_len - SAMPLE_RATE)
        waveform = waveform[:, start:start + SAMPLE_RATE]
    else:
        length = waveform.shape[1]
        if length < SAMPLE_RATE:
            waveform = F.pad(waveform, (0, SAMPLE_RATE - length))
        elif length > SAMPLE_RATE:
            waveform = waveform[:, :SAMPLE_RATE]
    return waveform

def to_log_mel(waveform):
    mel = mel_transform(waveform)
    return torch.log(mel + 1e-6)

class SpeechCommandsDataset(Dataset):
    def __init__(self, rows):
        self.rows = rows
    def __len__(self):
        return len(self.rows)
    def __getitem__(self, idx):

        spec = to_log_mel(load_fixed_waveform(path, label))
        return spec, label_to_idx[label]

    def __init__(self, num_classes=12):
        super().__init__()
        self.conv = nn.Sequential(
            nn.Conv2d(1, 16, kernel_size=3, padding=1), nn.BatchNorm2d(16), nn.ReLU(), nn.MaxPool2d(2),
            nn.Conv2d(16, 32, kernel_size=3, padding=1), nn.BatchNorm2d(32), nn.ReLU(), nn.MaxPool2d(2),
            nn.Conv2d(32, 64, kernel_size=3, padding=1), nn.BatchNorm2d(64), nn.ReLU(), nn.MaxPool2d(2),
        )
        self.classifier = nn.Sequential(
            nn.Flatten(),
            nn.Linear(64 * 5 * 12, 128), nn.ReLU(), nn.Dropout(0.3),
            nn.Linear(128, num_classes),
        )
    def forward(self, x):
        return self.classifier(self.conv(x))
